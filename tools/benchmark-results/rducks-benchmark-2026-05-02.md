# Scope

This benchmark exercises the real Rducks public path:

- R package registration through `rducks_register()`.
- DuckDB scalar-function execution on real DuckDB chunks.
- `eval_mode = "R"` and `eval_mode = "RC"` with `side_effects = TRUE`.
- Direct-buffer RC paths for ordinary scalar types.
- Arrow/nanoarrow fallback paths for exact and nested types.
- TPC-H data generated by DuckDB’s `tpch` extension when available.

It is deliberately **not** a package test. It is a profiling and
regression suite for local performance work. Defaults are small enough
for development; increase `params$sf`, `params$reps`, `params$micro_n`,
and `params$leak_iters` for serious runs.

Example:

``` sh
Rscript -e 'rmarkdown::render(
  "tools/benchmark_rducks.Rmd",
  output_dir = "bench-results",
  params = list(sf = 0.1, reps = 7, micro_n = c(1e5, 1e6, 3e6), leak_iters = 100)
)'
```

For native leak checking, render a tiny run once, then run the same
workload under Valgrind or an ASAN-built R/DuckDB/Rducks stack. The
in-process memory probes below are trend detectors, not proof of leak
freedom.

``` r
suppressPackageStartupMessages({
  library(DBI)
  library(duckdb)

bench_started <- Sys.time()

sql_string <- function(x) {
  paste0("'", gsub("'", "''", x, fixed = TRUE), "'")
}

rss_mb <- function() {
  path <- "/proc/self/status"
  if (!file.exists(path)) return(NA_real_)
  line <- grep("^VmRSS:", readLines(path, warn = FALSE), value = TRUE)
  if (!length(line)) return(NA_real_)
  as.numeric(sub("^VmRSS:[[:space:]]*([0-9]+)[[:space:]]*kB.*$", "\\1", line)) / 1024
}

gc_snapshot <- function() {
  g <- gc()
  c(
    ncells_used = unname(g["Ncells", "used"]),
    vcells_used = unname(g["Vcells", "used"])
  )
}

duckdb_memory_mb <- function(con) {
  tryCatch({
    x <- DBI::dbGetQuery(con, "SELECT sum(memory_usage_bytes) AS bytes FROM duckdb_memory()")
    as.numeric(x$bytes[[1]]) / 1024^2
  }, error = function(e) NA_real_)
}

memory_snapshot <- function(con) {
  g <- gc_snapshot()
  data.frame(
    rss_mb = rss_mb(),
    duckdb_memory_mb = duckdb_memory_mb(con),
    ncells_used = g[["ncells_used"]],
    vcells_used = g[["vcells_used"]],
    stringsAsFactors = FALSE
  )
}

connect_rducks <- function() {
  con <- DBI::dbConnect(duckdb::duckdb(config = list(allow_unsigned_extensions = "true")))
  # The benchmark intentionally exercises the supported calling-R-thread path.
  # Set these explicitly here as benchmark setup, not as hidden package behavior.
  DBI::dbExecute(con, "SET external_threads=1")
  DBI::dbExecute(con, "PRAGMA threads=1")
  rducks_enable(con, threads = "single")
  DBI::dbExecute(con, "SET external_threads=1")
  DBI::dbExecute(con, "PRAGMA threads=1")
  con
}

disconnect_rducks <- function(con) {
  try(DBI::dbDisconnect(con, shutdown = TRUE), silent = TRUE)
  invisible(NULL)
}

bench_sql <- function(con, label, sql, reps = params$reps, warmup = params$warmup) {
  for (i in seq_len(warmup)) invisible(DBI::dbGetQuery(con, sql))
  rows <- vector("list", reps)
  for (i in seq_len(reps)) {
    invisible(gc())
    before <- memory_snapshot(con)
    timing <- system.time(result <- DBI::dbGetQuery(con, sql))
    after_query <- memory_snapshot(con)
    invisible(gc())
    after_gc <- memory_snapshot(con)
    rows[[i]] <- data.frame(
      label = label,
      rep = i,
      elapsed_sec = unname(timing[["elapsed"]]),
      user_sec = unname(timing[["user.self"]]),
      system_sec = unname(timing[["sys.self"]]),
      result_rows = NROW(result),
      rss_before_mb = before$rss_mb,
      rss_after_query_mb = after_query$rss_mb,
      rss_after_gc_mb = after_gc$rss_mb,
      duckdb_memory_before_mb = before$duckdb_memory_mb,
      duckdb_memory_after_query_mb = after_query$duckdb_memory_mb,
      duckdb_memory_after_gc_mb = after_gc$duckdb_memory_mb,
      ncells_before = before$ncells_used,
      ncells_after_gc = after_gc$ncells_used,
      vcells_before = before$vcells_used,
      vcells_after_gc = after_gc$vcells_used,
      stringsAsFactors = FALSE
    )
  }
  do.call(rbind, rows)
}

summarise_bench <- function(x) {
  if (!NROW(x)) return(x)
  labels <- unique(x$label)
  do.call(rbind, lapply(labels, function(label) {
    y <- x[x$label == label, , drop = FALSE]
    data.frame(
      label = label,
      reps = NROW(y),
      min_sec = min(y$elapsed_sec),
      median_sec = median(y$elapsed_sec),
      mean_sec = mean(y$elapsed_sec),
      max_sec = max(y$elapsed_sec),
      rss_delta_after_gc_mb = tail(y$rss_after_gc_mb, 1) - y$rss_before_mb[[1]],
      duckdb_mem_delta_after_gc_mb = tail(y$duckdb_memory_after_gc_mb, 1) - y$duckdb_memory_before_mb[[1]],
      stringsAsFactors = FALSE
    )
  }))
}

write_bench <- function(x, name) {
  path <- file.path(bench_dir, paste0(name, ".csv"))
  utils::write.csv(x, path, row.names = FALSE)
  path
}

run_rprof <- function(con, label, sql, interval = 0.001) {
  prof_file <- file.path(bench_dir, paste0(label, ".Rprof"))
  utils::Rprof(prof_file, interval = interval, memory.profiling = TRUE)
  on.exit(utils::Rprof(NULL), add = TRUE)
  invisible(DBI::dbGetQuery(con, sql))
  utils::Rprof(NULL)
  summary <- tryCatch(
    utils::summaryRprof(prof_file, memory = "both"),
    error = function(e) utils::summaryRprof(prof_file)
  )
  by_self <- as.data.frame(summary$by.self)
  by_self$function_name <- rownames(by_self)
  rownames(by_self) <- NULL
  by_self <- by_self[, c("function_name", setdiff(names(by_self), "function_name")), drop = FALSE]
  utils::write.csv(by_self, file.path(bench_dir, paste0(label, "-rprof-by-self.csv")), row.names = FALSE)
  by_self
}

run_duckdb_json_profile <- function(con, label, sql) {
  path <- file.path(bench_dir, paste0(label, ".duckdb-profile.json"))
  DBI::dbExecute(con, paste0("PRAGMA profiling_output=", sql_string(normalizePath(path, mustWork = FALSE))))

  invisible(DBI::dbGetQuery(con, sql))
  DBI::dbExecute(con, "PRAGMA disable_profiling")
  path
}

  path
}
```

# Buffer-reuse semantic guard

Before looking for faster buffer reuse, preserve the semantic boundary:
Rducks may reuse DuckDB-owned input and output buffers inside one DuckDB
scalar-function callback, but it must not reuse mutable R argument
objects in a way that changes objects retained by user code. This guard
stores the first argument and first `sys.call()` from a UDF invocation
and checks that later rows do not mutate them.

``` r
con_guard <- connect_rducks()

retained <- new.env(parent = emptyenv())
retained$arg <- NULL
retained$call <- NULL

invisible(rducks_register(
  con_guard,
  "rd_reuse_guard_rc",
  function(x) {
    if (is.null(retained$arg)) retained$arg <- x
    if (is.null(retained$call)) retained$call <- sys.call()
    x + 1L
  },
  INTEGER,
  INTEGER,
  eval_mode = "RC",
  side_effects = TRUE
))

guard_result <- DBI::dbGetQuery(con_guard, "SELECT rd_reuse_guard_rc(i::INTEGER) AS x FROM range(5) t(i)")
reuse_guard <- data.frame(
  output = paste(guard_result$x, collapse = ","),
  retained_arg = retained$arg,
  retained_arg_ok = identical(retained$arg, 0L),
  retained_call = paste(deparse(retained$call), collapse = " "),
  stringsAsFactors = FALSE
)
write_bench(reuse_guard, "buffer-reuse-guard")
#> [1] "bench-results/rducks-20260502-193443/buffer-reuse-guard.csv"
reuse_guard
#>      output retained_arg retained_arg_ok
#> 1 1,2,3,4,5            0            TRUE
#>                                                                                                                                                       retained_call
#> 1 (function (x)  {     if (is.null(retained$arg))          retained$arg <- x     if (is.null(retained$call))          retained$call <- sys.call()     x + 1L })(0L)
```

# Microbenchmarks

The microbenchmarks isolate scalar UDF overhead from table scan
complexity. They include native SQL baselines, direct-buffer RC cases,
and fallback Arrow/nanoarrow cases.

# TPC-H-inspired benchmark

This section uses DuckDB `dbgen` tables and TPC-H Q1-style aggregations.
It is not a full audited TPC-H submission; it is a realistic row-heavy
workload for Rducks scalar UDF execution.

``` r
con_tpch <- connect_rducks()

ensure_tpch <- function(con) {
  loaded <- tryCatch({ DBI::dbExecute(con, "LOAD tpch"); TRUE }, error = function(e) FALSE)
  if (!loaded) {
    loaded <- tryCatch({
      DBI::dbExecute(con, "INSTALL tpch")
      DBI::dbExecute(con, "LOAD tpch")
      TRUE
    }, error = function(e) {
      warning("DuckDB tpch extension unavailable: ", conditionMessage(e), call. = FALSE)
      FALSE
    })
  }
  loaded
}

tpch_available <- ensure_tpch(con_tpch)
if (tpch_available) {
  DBI::dbExecute(con_tpch, sprintf("CALL dbgen(sf=%s)", as.character(params$sf)))

    data.frame(table = "orders", rows = DBI::dbGetQuery(con_tpch, "SELECT count(*) AS n FROM orders")$n),
    data.frame(table = "customer", rows = DBI::dbGetQuery(con_tpch, "SELECT count(*) AS n FROM customer")$n)
  )
  write_bench(tpch_counts, "tpch-counts")
```

# Memory trend probes

These loops repeat already-registered UDFs. They are intended to catch
obvious per-call leaks in the execution path. They do not treat one-time
registration state as a leak, and they do not prove allocator memory
must return to the OS. Inspect the slope and the raw CSV in `bench_dir`.

``` r
con_leak <- connect_rducks()

invisible(rducks_register(con_leak, "rd_leak_double_rc", function(x) x + 1, DOUBLE, DOUBLE,
                          eval_mode = "RC", side_effects = TRUE))
invisible(rducks_register(con_leak, "rd_leak_decimal_rc", function(x) x + rducks_decimal("1.00", 15, 2), DECIMAL(15, 2), DECIMAL(15, 2),
                          eval_mode = "RC", side_effects = TRUE))
invisible(rducks_register(con_leak, "rd_leak_struct_rc", function(x) x$a + x$b, STRUCT(a = INTEGER, b = INTEGER), INTEGER,
                          eval_mode = "RC", side_effects = TRUE))

leak_probe <- function(con, label, sql, iterations) {
  rows <- vector("list", iterations)
  invisible(DBI::dbGetQuery(con, sql))
  for (i in seq_len(iterations)) {
    invisible(gc())
    before <- memory_snapshot(con)
    timing <- system.time(invisible(DBI::dbGetQuery(con, sql)))
    invisible(gc())
    after <- memory_snapshot(con)
    rows[[i]] <- data.frame(
      label = label,
      iter = i,
      elapsed_sec = unname(timing[["elapsed"]]),
      rss_before_mb = before$rss_mb,
      rss_after_gc_mb = after$rss_mb,
      duckdb_memory_before_mb = before$duckdb_memory_mb,
      duckdb_memory_after_gc_mb = after$duckdb_memory_mb,
      ncells_before = before$ncells_used,
      ncells_after_gc = after$ncells_used,
      vcells_before = before$vcells_used,
      vcells_after_gc = after$vcells_used,
      stringsAsFactors = FALSE
    )
  }
  do.call(rbind, rows)
}

leak_n <- max(10000L, min(as.integer(unlist(params$micro_n)), na.rm = TRUE))
leak_queries <- data.frame(
  label = c("leak/RC-double-direct", "leak/RC-decimal-fallback", "leak/RC-struct-fallback"),
  sql = c(
    sprintf("SELECT sum(rd_leak_double_rc(i::DOUBLE)) AS x FROM range(%d) t(i)", leak_n),
    sprintf("SELECT sum(rd_leak_decimal_rc((i %% 1000)::DECIMAL(15,2))::DOUBLE) AS x FROM range(%d) t(i)", leak_n),
    sprintf("SELECT sum(rd_leak_struct_rc({'a': i::INTEGER, 'b': 1::INTEGER}::STRUCT(a INTEGER, b INTEGER))) AS x FROM range(%d) t(i)", leak_n)
  ),
  stringsAsFactors = FALSE
)

leak_results <- do.call(rbind, lapply(seq_len(NROW(leak_queries)), function(i) {
  leak_probe(con_leak, leak_queries$label[[i]], leak_queries$sql[[i]], params$leak_iters)
}))

leak_slopes <- do.call(rbind, lapply(split(leak_results, leak_results$label), function(x) {
  data.frame(
    label = x$label[[1]],
    iterations = NROW(x),
    rss_slope_mb_per_iter = unname(coef(lm(rss_after_gc_mb ~ iter, data = x))[["iter"]]),

}))
```
